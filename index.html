<script>
  // ---- WHY THIS IS BUILT THIS WAY --------------------------------------
  // The single-threaded @ffmpeg/core cannot run two commands. When an
  // ffmpeg command finishes, ffmpeg calls exit(), which aborts the whole
  // wasm runtime -- the diagnostic log shows "Aborted()" at the end of
  // every completed run. Anything executed afterwards runs on a dead core
  // and returns nothing, which surfaced as the misleading errors
  // "No video stream found" and "Encoding produced an empty file".
  //
  // So: one fresh core per ffmpeg command.
  //   phase 1  probe   -> read stream details, then terminate the core
  //   phase 2  encode   -> a brand new core, with shrink's internal probe
  //                        stubbed out so the encode is its ONLY command
  //
  // capabilities() would also spend a command on `ffmpeg -encoders`, so
  // _caps is pre-seeded from this core's real encoder list.
  var CAPS = {
    h264: true, vp9: true, vp8: true, hevc: true, av1: false,
    aac: true, opus: true, multithread: false,
    all: { libx264:true, 'libvpx-vp9':true, libvpx:true, libx265:true,
           aac:true, libopus:true, libvorbis:true }
  };

  function makeEngine(onProgress) {
    var e = new Scaffold({ corePath: 'tiny6', coreType: 'st' });
    e._caps = CAPS;
    if (onProgress) e._prog = onProgress;
    return e;
  }

  // Probe on one core, encode on another. The core is cheap to recreate
  // because the 31 MB wasm is already in the browser cache.
  function convert(file, opts) {
    var prog = opts.onProgress;
    var probeEngine = makeEngine(prog);
    return probeEngine.probe(file).then(function (info) {
      probeEngine.terminate();
      var encEngine = makeEngine(prog);
      // shrink() calls this.probe() internally; that would burn this
      // core's one command before the encode ever starts.
      encEngine.probe = function () { return Promise.resolve(info); };
      return encEngine.shrink(file, opts).then(function (r) {
        encEngine.terminate();
        return r;
      }, function (err) {
        encEngine.terminate();
        throw err;
      });
    }, function (err) {
      probeEngine.terminate();
      throw err;
    });
  }

  var chosen = null, result = null;

  // ---- file selection --------------------------------------------------
  var drop = $('drop'), input = $('file');
  drop.addEventListener('click', function () { input.click(); });
  drop.addEventListener('dragover', function (e) { e.preventDefault(); drop.classList.add('over'); });
  drop.addEventListener('dragleave', function () { drop.classList.remove('over'); });
  drop.addEventListener('drop', function (e) {
    e.preventDefault(); drop.classList.remove('over');
    if (e.dataTransfer.files.length) pick(e.dataTransfer.files[0]);
  });
  input.addEventListener('change', function () { if (input.files.length) pick(input.files[0]); });

  function pick(f) {
    chosen = f;
    drop.textContent = f.name + '  ·  ' + (f.size / 1048576).toFixed(2) + ' MB';
    drop.appendChild(input);
    $('run').disabled = false;
    $('stat').textContent = 'Ready to convert.';
  }

  // ---- controls --------------------------------------------------------
  $('crf').addEventListener('input', function () { $('crfV').textContent = this.value; });
  $('luma').addEventListener('input', function () {
    var L = +this.value / 100;
    if (L >= 1) {
      $('lumaV').textContent = 'off';
      $('lumaHint').textContent = 'Luma is off. At matched size, raising CRF beat luma scaling on every clip tested.';
    } else {
      $('lumaV').textContent = L.toFixed(2);
      $('lumaHint').textContent = 'White 235 \u2192 ' + Math.round((235 - 16) * L + 16) +
        '. This is permanent; no player undoes it.';
    }
  });

  // ---- convert ---------------------------------------------------------
  $('run').addEventListener('click', function () {
    if (!chosen) return;
    $('run').disabled = true;
    $('save').classList.add('hide');
    $('outPanel').classList.add('hide');
    $('stat').textContent = 'Loading ffmpeg\u2026';

    var opts = Scaffold.preset('tn6', {
      crf: +$('crf').value,
      luma: (+$('luma').value) / 100,
      onProgress: function (p) {
        $('barI').style.width = p.percent + '%';
        $('stat').textContent = p.message +
          (p.stage === 'encode' && p.percent > 0 ? ' \u2014 ' + Math.round(p.percent) + '%' : '');
      }
    });

    convert(chosen, opts).then(function (r) {
      result = r;
      $('barI').style.width = '100%';
      $('stat').textContent = r.stats.summary;
      $('save').classList.remove('hide');
      $('outPanel').classList.remove('hide');
      return Tiny6Player.attach($('player'), r.blob);
    }).then(function (meta) {
      var s = result.stats;
      $('meta').innerHTML =
        row('file', result.filename) +
        row('size', fmt(s.originalBytes) + ' \u2192 ' + fmt(s.outputBytes) +
                    '  (' + (s.grew ? '+' : '\u2212') + Math.abs(s.reductionPercent).toFixed(1) + '%)') +
        row('coded', meta.codedWidth + '\u00d7' + meta.codedHeight) +
        row('display', meta.displayWidth + '\u00d7' + meta.displayHeight + ' (anamorphic)') +
        row('codec', meta.videoCodec + (meta.audioCodec ? ' / ' + meta.audioCodec : '')) +
        row('crf', s.crf) +
        (s.luma ? row('luma', s.luma.toFixed(2) + ' (' + s.lumaMode + ')') : '') +
        (s.ssim != null ? row('ssim', s.ssim.toFixed(4)) : '');
      $('warns').innerHTML = (s.warnings || []).map(function (w) {
        return '
' + esc(w) + '
';
      }).join('');
      $('run').disabled = false;
    })['catch'](function (err) {
      $('stat').textContent = err.message;
      $('run').disabled = false;
    });
  });

  $('save').addEventListener('click', function () {
    if (!result) return;
    var a = document.createElement('a');
    a.href = URL.createObjectURL(result.blob);
    a.download = result.filename;
    a.click();
    setTimeout(function () { URL.revokeObjectURL(a.href); }, 4000);
  });

  function row(k, v) { return '
' + k + '
' + esc(String(v)) + '
'; }
  function esc(s) { return s.replace(/[&<>]/g, function (c) {
    return { '&': '&', '<': '<', '>': '>' }[c]; }); }
  function fmt(n) {
    return n >= 1048576 ? (n / 1048576).toFixed(2) + ' MB' : (n / 1024).toFixed(1) + ' KB';
  }
})();
