<!-- CONVERTING - needs ffmpeg.wasm, ~31 MB. Same origin, not a CDN:
     @ffmpeg/ffmpeg builds a Worker and cross-origin workers are blocked.
     814.ffmpeg.js is not referenced anywhere but IS required - ffmpeg.js
     loads it as its worker, from its own folder. Ship all five. -->
<script src="tiny6/ffmpeg.js"></script>
<script src="tiny6/util.js"></script>
<script src="tiny6/scaffold.js"></script>
<script src="tiny6/tiny6-player.js"></script>
<!-- plus tiny6/814.ffmpeg.js, ffmpeg-core.js, ffmpeg-core.wasm -->

<script>
  // This core serves ONE ffmpeg command per instance, so build a fresh
  // engine for each step rather than reusing one.
  function engine() {
    var e = new Scaffold({ corePath: 'tiny6', coreType: 'st' });
    // Skips `ffmpeg -encoders`, which would spend the core's one command.
    e._caps = { h264:true, vp9:true, vp8:true, hevc:true, av1:false,
                aac:true, opus:true, multithread:false, all:{} };
    return e;
  }

  // NOTE: preset('tn6') defaults to VP9, and the libvpx-vp9 encoder traps
  // in this core build. Use VP8.
  var e1 = engine();
  e1.probe(file).then(function (info) {
    e1.terminate();
    var e2 = engine();
    e2.probe = function () { return Promise.resolve(info); };
    return e2.shrink(file, Scaffold.preset('tn6', { codec: 'vp8' }))
             .then(function (r) { e2.terminate(); return r; });
  }).then(function (r) {
    Tiny6Player.attach(videoEl, r.blob);
  });
</script>

scaffold.js converts. tiny6-player.js plays, and auto-upgrades any <video data-tn6="…"> on load. The player has no dependencies at all — a page that only plays .tn6 files ships one 6 KB file instead of ~31 MB of ffmpeg.wasm.
